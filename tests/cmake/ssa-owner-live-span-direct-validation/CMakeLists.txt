cmake_minimum_required(VERSION 3.31)
# Configure only this actual checkout; no source copies or downloads.
get_filename_component(_zr_live_span_actual_root "${CMAKE_CURRENT_LIST_DIR}/../../.." REALPATH)
cmake_path(CONVERT "E:/Git/zr_vm" TO_CMAKE_PATH_LIST _zr_live_span_expected_root NORMALIZE)
if (NOT _zr_live_span_actual_root STREQUAL _zr_live_span_expected_root)
    message(FATAL_ERROR "Finite live-span validation requires the actual E:/Git/zr_vm checkout")
endif ()
set(ZR_SSA_OWNER_LIVE_SPAN_SOURCE_ROOT "${_zr_live_span_actual_root}")
set(_zr_live_span_output_root "E:/cargo-targets/zr_vm/build")
cmake_path(ABSOLUTE_PATH CMAKE_BINARY_DIR NORMALIZE OUTPUT_VARIABLE _zr_live_span_binary)
cmake_path(IS_PREFIX _zr_live_span_output_root "${_zr_live_span_binary}"
        NORMALIZE _zr_live_span_output_allowed)
if (NOT _zr_live_span_output_allowed OR _zr_live_span_binary STREQUAL _zr_live_span_output_root
        OR CMAKE_BINARY_DIR STREQUAL CMAKE_SOURCE_DIR)
    message(FATAL_ERROR "Use a separate child directory under E:/cargo-targets/zr_vm/build")
endif ()
if (DEFINED CMAKE_TRY_COMPILE_TARGET_TYPE AND
        NOT CMAKE_TRY_COMPILE_TARGET_TYPE STREQUAL "EXECUTABLE")
    message(FATAL_ERROR "Use the actual executable compiler ABI probe")
endif ()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
project(zr_ssa_owner_live_span_validation LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ZR_SSA_OWNER_LIVE_SPAN_UBSAN "Enable finite Clang UBSan with no recovery" ON)
if (ZR_SSA_OWNER_LIVE_SPAN_UBSAN AND CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    foreach (_flag IN ITEMS "/clang:-fsanitize=undefined" "/clang:-fno-sanitize-recover=all")
        string(FIND "${CMAKE_C_LINK_EXECUTABLE}" "${_flag}" _flag_position)
        string(FIND "${CMAKE_C_LINK_EXECUTABLE}" "/link" _link_position)
        if (_flag_position LESS 0 OR _link_position LESS 0 OR NOT _flag_position LESS _link_position)
            message(FATAL_ERROR "Toolchain must pass UBSan driver flags before /link")
        endif ()
    endforeach ()
endif ()
enable_testing()
include("${ZR_SSA_OWNER_LIVE_SPAN_SOURCE_ROOT}/tests/cmake/ssa-owner-live-span.cmake")
