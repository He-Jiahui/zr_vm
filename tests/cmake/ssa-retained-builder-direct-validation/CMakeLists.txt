cmake_minimum_required(VERSION 3.31)

include("${CMAKE_CURRENT_LIST_DIR}/../ssa-direct-validation-paths.cmake")
zr_ssa_validation_source_root(_zr_direct_source_root)
set(ZR_SSA_RETAINED_BUILDER_SOURCE_ROOT "${_zr_direct_source_root}")
zr_ssa_validation_binary_dir(_zr_direct_binary_dir)
if (DEFINED CMAKE_TRY_COMPILE_TARGET_TYPE AND NOT CMAKE_TRY_COMPILE_TARGET_TYPE STREQUAL "EXECUTABLE")
    message(FATAL_ERROR "Use the actual executable compiler ABI probe")
endif ()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
project(zr_ssa_retained_builder_validation LANGUAGES C)
if (NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja" OR NOT CMAKE_BUILD_TYPE STREQUAL "Debug"
    OR NOT CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    message(FATAL_ERROR "Finite retained builder route requires Windows MSVC-frontend/Ninja/Debug")
endif ()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ZR_SSA_RETAINED_BUILDER_UBSAN "Enable finite Clang UBSan with no recovery" ON)
if (ZR_SSA_RETAINED_BUILDER_UBSAN)
    if (NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
        message(FATAL_ERROR "Finite retained builder UBSan requires Clang")
    endif ()
    foreach (_flag IN ITEMS "/clang:-fsanitize=undefined" "/clang:-fno-sanitize-recover=all")
        string(FIND "${CMAKE_C_LINK_EXECUTABLE}" "${_flag}" _flag_position)
        string(FIND "${CMAKE_C_LINK_EXECUTABLE}" "/link" _link_position)
        if (_flag_position LESS 0 OR _link_position LESS 0 OR NOT _flag_position LESS _link_position)
            message(FATAL_ERROR "Toolchain must pass UBSan driver flags before /link")
        endif ()
    endforeach ()
endif ()
enable_testing()
include("${ZR_SSA_RETAINED_BUILDER_SOURCE_ROOT}/tests/cmake/ssa-retained-builder-cfg.cmake")
