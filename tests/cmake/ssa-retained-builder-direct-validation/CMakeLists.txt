cmake_minimum_required(VERSION 3.31)
# Configure actual current sources. Root reviews the closure before execution.
get_filename_component(_zr_retained_cfg_actual_root "${CMAKE_CURRENT_LIST_DIR}/../../.." REALPATH)
cmake_path(CONVERT "E:/Git/zr_vm" TO_CMAKE_PATH_LIST _zr_retained_cfg_expected_root NORMALIZE)
if (NOT _zr_retained_cfg_actual_root STREQUAL _zr_retained_cfg_expected_root)
    message(FATAL_ERROR "Finite retained builder validation requires actual E:/Git/zr_vm")
endif ()
set(ZR_SSA_RETAINED_BUILDER_SOURCE_ROOT "${_zr_retained_cfg_actual_root}")
set(_zr_retained_cfg_output_root "E:/cargo-targets/zr_vm/build")
cmake_path(ABSOLUTE_PATH CMAKE_BINARY_DIR NORMALIZE OUTPUT_VARIABLE _zr_retained_cfg_binary)
cmake_path(IS_PREFIX _zr_retained_cfg_output_root "${_zr_retained_cfg_binary}"
    NORMALIZE _zr_retained_cfg_output_allowed)
if (NOT _zr_retained_cfg_output_allowed OR _zr_retained_cfg_binary STREQUAL _zr_retained_cfg_output_root
    OR CMAKE_BINARY_DIR STREQUAL CMAKE_SOURCE_DIR)
    message(FATAL_ERROR "Use a separate child directory under E:/cargo-targets/zr_vm/build")
endif ()
if (DEFINED CMAKE_TRY_COMPILE_TARGET_TYPE AND NOT CMAKE_TRY_COMPILE_TARGET_TYPE STREQUAL "EXECUTABLE")
    message(FATAL_ERROR "Use the actual executable compiler ABI probe")
endif ()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
project(zr_ssa_retained_builder_validation LANGUAGES C)
if (NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja" OR NOT CMAKE_BUILD_TYPE STREQUAL "Debug"
    OR NOT CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    message(FATAL_ERROR "Finite retained builder route requires Windows MSVC-frontend/Ninja/Debug")
endif ()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ZR_SSA_RETAINED_BUILDER_UBSAN "Enable finite Clang UBSan with no recovery" ON)
if (ZR_SSA_RETAINED_BUILDER_UBSAN)
    if (NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
        message(FATAL_ERROR "Finite retained builder UBSan requires Clang")
    endif ()
    foreach (_flag IN ITEMS "/clang:-fsanitize=undefined" "/clang:-fno-sanitize-recover=all")
        string(FIND "${CMAKE_C_LINK_EXECUTABLE}" "${_flag}" _flag_position)
        string(FIND "${CMAKE_C_LINK_EXECUTABLE}" "/link" _link_position)
        if (_flag_position LESS 0 OR _link_position LESS 0 OR NOT _flag_position LESS _link_position)
            message(FATAL_ERROR "Toolchain must pass UBSan driver flags before /link")
        endif ()
    endforeach ()
endif ()
enable_testing()
include("${ZR_SSA_RETAINED_BUILDER_SOURCE_ROOT}/tests/cmake/ssa-retained-builder-cfg.cmake")
